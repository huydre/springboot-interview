OrderState.java
enum OrderState {
  PENDING { OrderState pay()  { return PAID; } },
  PAID    { OrderState ship() { return SHIPPED; } },
  SHIPPED { };
  OrderState pay()  { throw new IllegalStateException(...); }
  OrderState ship() { throw new IllegalStateException(...); }
✓ PENDING → PAID → SHIPPED
✕ PENDING.ship() ném: cannot ship in PENDING
✕ SHIPPED.pay() ném: cannot pay in SHIPPED
